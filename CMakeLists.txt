cmake_minimum_required(VERSION 3.24)
project(HdrBridge VERSION 0.3.4 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use Windows 11 + Visual Studio 2022 (C++ desktop workload and Windows SDK).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/color.hlsl" COLOR_SHADER)
configure_file(src/shader_source.h.in generated/shader_source.h @ONLY)
add_library(bridge_options INTERFACE)
target_include_directories(bridge_options INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(bridge_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(bridge_options INTERFACE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(bridge_options INTERFACE d3d11 dxgi d3dcompiler windowsapp user32 gdi32 shell32 comctl32 dwmapi ole32 mmdevapi uuid)
add_executable(HdrBridge WIN32 src/main.cpp src/audio_bridge.cpp)
target_link_libraries(HdrBridge PRIVATE bridge_options)
enable_testing()
add_executable(ColorPipelineTests tests/color_pipeline.cpp)
target_link_libraries(ColorPipelineTests PRIVATE bridge_options)
add_test(NAME color_pipeline COMMAND ColorPipelineTests)
add_executable(StreamSettingsTests tests/stream_settings.cpp)
target_link_libraries(StreamSettingsTests PRIVATE bridge_options)
add_test(NAME stream_settings COMMAND StreamSettingsTests)
add_executable(PreviewPipelineTests tests/preview_pipeline.cpp)
target_link_libraries(PreviewPipelineTests PRIVATE bridge_options)
add_test(NAME preview_pipeline COMMAND PreviewPipelineTests)
add_executable(AudioBufferTests tests/audio_buffer.cpp)
target_link_libraries(AudioBufferTests PRIVATE bridge_options)
add_test(NAME audio_buffer COMMAND AudioBufferTests)
install(TARGETS HdrBridge RUNTIME DESTINATION .)
