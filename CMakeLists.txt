cmake_minimum_required(VERSION 3.24)
project(HdrBridge VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use Windows 11 + Visual Studio 2022 (C++ desktop workload and Windows SDK).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/shaders/color.hlsl" COLOR_SHADER)
configure_file(src/shader_source.h.in generated/shader_source.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ui/index.html ui/theme.css ui/app.js)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ui/index.html" UI_HTML_TEMPLATE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ui/theme.css" UI_CSS)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ui/app.js" UI_JS)
string(CONFIGURE "${UI_HTML_TEMPLATE}" UI_HTML @ONLY)
configure_file(src/ui_source.h.in generated/ui_source.h @ONLY)

set(WEBVIEW2_VERSION "1.0.3537.50")
set(WEBVIEW2_HASH "5ea526bbd728adda0da4d31219267e96460494a427e4894c4e09d9f320f4b9aa")
set(WEBVIEW2_ROOT "${CMAKE_CURRENT_BINARY_DIR}/webview2/${WEBVIEW2_VERSION}")
set(WEBVIEW2_ARCHIVE "${CMAKE_CURRENT_BINARY_DIR}/webview2-${WEBVIEW2_VERSION}.nupkg")
file(DOWNLOAD "https://api.nuget.org/v3-flatcontainer/microsoft.web.webview2/${WEBVIEW2_VERSION}/microsoft.web.webview2.${WEBVIEW2_VERSION}.nupkg"
  "${WEBVIEW2_ARCHIVE}" TLS_VERIFY ON EXPECTED_HASH "SHA256=${WEBVIEW2_HASH}" TIMEOUT 90)
if(NOT EXISTS "${WEBVIEW2_ROOT}/build/native/include/WebView2.h")
  file(ARCHIVE_EXTRACT INPUT "${WEBVIEW2_ARCHIVE}" DESTINATION "${WEBVIEW2_ROOT}")
endif()
configure_file("${WEBVIEW2_ROOT}/LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/WebView2-LICENSE.txt" COPYONLY)
if(CMAKE_VS_PLATFORM_NAME STREQUAL "ARM64")
  set(WEBVIEW2_ARCH arm64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(WEBVIEW2_ARCH x64)
else()
  set(WEBVIEW2_ARCH x86)
endif()
add_library(bridge_options INTERFACE)
target_include_directories(bridge_options INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(bridge_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(bridge_options INTERFACE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(bridge_options INTERFACE d3d11 dxgi d3dcompiler windowsapp user32 gdi32 shell32 comctl32 dwmapi ole32 mmdevapi uuid)
add_executable(HdrBridge WIN32 src/main.cpp src/audio_bridge.cpp)
target_include_directories(HdrBridge SYSTEM PRIVATE "${WEBVIEW2_ROOT}/build/native/include")
target_link_libraries(HdrBridge PRIVATE bridge_options advapi32 shlwapi version
  "${WEBVIEW2_ROOT}/build/native/${WEBVIEW2_ARCH}/WebView2LoaderStatic.lib")
enable_testing()
add_executable(ColorPipelineTests tests/color_pipeline.cpp)
target_link_libraries(ColorPipelineTests PRIVATE bridge_options)
add_test(NAME color_pipeline COMMAND ColorPipelineTests)
add_executable(StreamSettingsTests tests/stream_settings.cpp)
target_link_libraries(StreamSettingsTests PRIVATE bridge_options)
add_test(NAME stream_settings COMMAND StreamSettingsTests)
add_executable(PreviewPipelineTests tests/preview_pipeline.cpp)
target_link_libraries(PreviewPipelineTests PRIVATE bridge_options)
add_test(NAME preview_pipeline COMMAND PreviewPipelineTests)
add_executable(AudioBufferTests tests/audio_buffer.cpp)
target_link_libraries(AudioBufferTests PRIVATE bridge_options)
add_test(NAME audio_buffer COMMAND AudioBufferTests)
install(TARGETS HdrBridge RUNTIME DESTINATION .)
install(FILES LICENSE "${CMAKE_CURRENT_BINARY_DIR}/WebView2-LICENSE.txt" DESTINATION .)
